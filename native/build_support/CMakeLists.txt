cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
project(MovingDeckLab LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(AIRSIMSGA_BACKEND "" CACHE PATH "Assembled moving-deck backend source")
set(AIRSIMSGA_SIMLIBS "" CACHE PATH "Prepared ProjectAirSim plugin SimLibs")
set(AIRSIMSGA_DEPS "" CACHE PATH "Dependency sources: msgpack-src, tinygltf-src, googletest-src")
foreach(required AIRSIMSGA_BACKEND AIRSIMSGA_SIMLIBS AIRSIMSGA_DEPS)
  if(NOT IS_DIRECTORY "${${required}}")
    message(FATAL_ERROR "Set ${required} to an existing external dependency directory")
  endif()
endforeach()
if(NOT MSVC)
  message(FATAL_ERROR "This component recipe follows the tested Windows/MSVC build; Linux native build not validated")
endif()
get_filename_component(RELEASE "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(SRC "${AIRSIMSGA_BACKEND}")
set(UE_PLUGIN_SIMLIBS_DIR "${AIRSIMSGA_SIMLIBS}")
set(SIMLIBS_TEST_DIR "${CMAKE_BINARY_DIR}/tests")
set(EIGEN_INCLUDE_DIR "${UE_PLUGIN_SIMLIBS_DIR}/eigen/include")
set(JSON_INCLUDE_DIR "${UE_PLUGIN_SIMLIBS_DIR}/json/include")
set(NNG_INCLUDE_DIR "${UE_PLUGIN_SIMLIBS_DIR}/nng/include")
set(NNG_LIB_DIR "${UE_PLUGIN_SIMLIBS_DIR}/nng/${CMAKE_BUILD_TYPE}")
set(MSGPACK_INCLUDE_DIR "${AIRSIMSGA_DEPS}/msgpack-src/include")
set(TINYGLTF_INCLUDE_DIR "${AIRSIMSGA_DEPS}/tinygltf-src")
set(ONNXRUNTIME_ROOTDIR "${UE_PLUGIN_SIMLIBS_DIR}/shared_libs/onnxruntime")
set(ONNX_LIB_DIR "${UE_PLUGIN_SIMLIBS_DIR}/shared_libs")
add_compile_options(/EHsc /bigobj)
add_compile_definitions(MSGPACK_NO_BOOST _SILENCE_CXX17_RESULT_OF_DEPRECATION_WARNING)
add_link_options(/MANIFEST:NO)
link_directories("${ONNX_LIB_DIR}" "${NNG_LIB_DIR}")
# Dependencies are caller-provided read-only inputs. Build outputs go to -B.
add_custom_target(nng-external)
add_subdirectory("${SRC}/tools/lvmon" lvmon)
add_subdirectory("${SRC}/core_sim/src" core_sim)
add_subdirectory("${SRC}/physics/src" physics)
add_subdirectory("${SRC}/mavlinkcom/src" mavlinkcom)
add_subdirectory("${SRC}/rendering/scene/src" rendering_scene)
add_subdirectory("${SRC}/vehicle_apis/multirotor_api/src" multirotor_api)
add_subdirectory("${SRC}/vehicle_apis/rover_api/src" rover_api)
add_subdirectory("${SRC}/simserver/src" simserver)
enable_testing()
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
add_subdirectory("${AIRSIMSGA_DEPS}/googletest-src" googletest EXCLUDE_FROM_ALL)
add_executable(support_contact_tests "${RELEASE}/tests/support_contact_tests.cpp")
target_include_directories(support_contact_tests PRIVATE "${SRC}/physics/include" "${SRC}/core_sim/include" "${EIGEN_INCLUDE_DIR}")
target_link_libraries(support_contact_tests PRIVATE gtest_main)
include(GoogleTest)
gtest_discover_tests(support_contact_tests)
add_executable(fastphysics_regression "${SRC}/physics/test/gtest_fast_physics.cpp"
    "${SRC}/physics/test/gtest_physics_world.cpp" "${SRC}/physics/test/gtest_unreal_physics.cpp")
target_include_directories(fastphysics_regression PRIVATE "${SRC}/physics/test"
    "${SRC}/physics/include" "${SRC}/core_sim/include" "${SRC}/core_sim/src"
    "${EIGEN_INCLUDE_DIR}" "${JSON_INCLUDE_DIR}" "${NNG_INCLUDE_DIR}" "${MSGPACK_INCLUDE_DIR}")
target_link_libraries(fastphysics_regression PRIVATE physics core_sim lvmon gtest_main)
add_custom_command(TARGET fastphysics_regression POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${ONNX_LIB_DIR}/onnxruntime.dll" "$<TARGET_FILE_DIR:fastphysics_regression>")
gtest_discover_tests(fastphysics_regression)

add_executable(firmware_disarm_tests "${RELEASE}/tests/firmware_disarm_tests.cpp")
target_include_directories(firmware_disarm_tests PRIVATE
    "${SRC}/vehicle_apis/multirotor_api/include" "${SRC}/vehicle_apis/include"
    "${SRC}/core_sim/include" "${EIGEN_INCLUDE_DIR}" "${JSON_INCLUDE_DIR}")
target_link_libraries(firmware_disarm_tests PRIVATE core_sim lvmon gtest_main)
add_custom_command(TARGET firmware_disarm_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${ONNX_LIB_DIR}/onnxruntime.dll" "$<TARGET_FILE_DIR:firmware_disarm_tests>")
gtest_discover_tests(firmware_disarm_tests)
